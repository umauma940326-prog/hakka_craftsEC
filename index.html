        // Tab Switching Logic

        function switchTab(tabId) {

            // Hide all tabs

            document.querySelectorAll('.tab-content').forEach(el => el.classList.remove('active'));

            // Reset nav colors

            document.querySelectorAll('.nav-item').forEach(el => {

                el.classList.remove('active');

                el.querySelector('i').classList.replace('text-gray-800', 'text-gray-400');

                el.querySelector('span').classList.replace('text-gray-900', 'text-gray-500');

            });

            

            // Show selected tab

            document.getElementById(tabId).classList.add('active');

            

            // Highlight selected nav

            const navId = tabId.replace('tab-', 'nav-');

            const activeNav = document.getElementById(navId);

            if(activeNav) {

                activeNav.classList.add('active');

                activeNav.querySelector('i').classList.replace('text-gray-400', 'text-gray-800');

                activeNav.querySelector('span').classList.replace('text-gray-500', 'text-gray-900');

            }


            // Close any open overlay pages when switching main tabs

            document.querySelectorAll('.overlay-view').forEach(el => el.classList.remove('show'));

        }


        // Initialize active state colors for Home on load

        document.getElementById('nav-home').querySelector('i').classList.replace('text-gray-400', 'text-gray-800');

        document.getElementById('nav-home').querySelector('span').classList.replace('text-gray-500', 'text-gray-900');
